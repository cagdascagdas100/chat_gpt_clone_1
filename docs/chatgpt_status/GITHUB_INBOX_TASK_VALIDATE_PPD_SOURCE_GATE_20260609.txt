$ErrorActionPreference='Continue'
$Repo='C:\Users\cagda\Documents\GitHub\AAYS'
$Branch='feature/terrayield-aays-integration'
$Work='F:\chatgpt\AAYS_WORK\source_generation_reports'
$Ppd='F:\sold_buildings\open_sources\ppd\pp-complete.csv'
$Incoming='F:\sold_buildings\licensed\incoming'
$RepoStatus=Join-Path $Repo 'docs\chatgpt_status'
New-Item -ItemType Directory -Force $Work,$Incoming,$RepoStatus | Out-Null
$FReport=Join-Path $Work 'AUTONOMOUS_PPD_SOURCE_GATE_LATEST.txt'
$GReport=Join-Path $RepoStatus 'AUTONOMOUS_PPD_SOURCE_GATE_LATEST.txt'
$exists=Test-Path $Ppd
$bytes=0; $sha=''; $first=''; $last=''; $sample=0; $ppdValid=$false; $warnings=@(); $blockers=@()
if($exists){
  $item=Get-Item $Ppd
  $bytes=$item.Length
  try { $sha=(Get-FileHash $Ppd -Algorithm SHA256).Hash } catch { $warnings+='sha256_failed' }
  try { $first=(Get-Content $Ppd -TotalCount 1 -ErrorAction Stop) } catch { $first='READ_FIRST_FAILED'; $warnings+='read_first_failed' }
  try { $last=(Get-Content $Ppd -Tail 1 -ErrorAction Stop) } catch { $last='READ_LAST_FAILED'; $warnings+='read_last_failed' }
  try { $sample=(Get-Content $Ppd -TotalCount 1000 -ErrorAction Stop | Measure-Object -Line).Lines } catch { $warnings+='sample_read_failed' }
  if($bytes -gt 4000000000 -and $first -notmatch '<html|<!DOCTYPE|AccessDenied|Error' -and $sample -gt 10){ $ppdValid=$true }
}else{ $blockers+='ppd_file_missing' }
if(-not $ppdValid){ $blockers+='ppd_not_validated' }
$files=Get-ChildItem $Incoming -File -ErrorAction SilentlyContinue
$hasAddress=(@($files | Where-Object { $_.Name -match 'addressbase|address_to_uprn' }).Count -gt 0)
$hasCombined=(@($files | Where-Object { $_.Name -match 'title_no_uprn_polygon_bridge|polygon_bridge' }).Count -gt 0)
$hasPolygon=(@($files | Where-Object { $_.Name -match 'national_polygon|polygon_bridge' }).Count -gt 0)
$hasTitle=(@($files | Where-Object { $_.Name -match 'title.*uprn|polygon_bridge' }).Count -gt 0)
$gateReady=($hasAddress -and ($hasCombined -or ($hasPolygon -and $hasTitle)))
if(-not $hasAddress){ $blockers+='required_address_to_uprn_missing' }
if(-not ($hasCombined -or ($hasPolygon -and $hasTitle))){ $blockers+='required_hmlr_nps_title_uprn_polygon_missing' }
if($ppdValid -and $gateReady){ $progress=75; $status='SOURCE_GATE_READY_FOR_V3'; $next='queue_full_db_address_match_v3_preflight'; $wait='5-15' }
elseif($ppdValid){ $progress=60; $status='PPD_VALID_SOURCE_GATE_BLOCKED'; $next='wait_for_licensed_sources'; $wait='0' }
else { $progress=55; $status='PPD_VALIDATION_FAILED'; $next='repair_or_redownload_ppd'; $wait='5-15' }
$lines=@(
  "status=$status",
  "updated_at=$(Get-Date -Format s)",
  "ppd_file=$Ppd",
  "ppd_exists=$exists",
  "ppd_valid=$ppdValid",
  "ppd_bytes=$bytes",
  "ppd_sha256=$sha",
  "ppd_first_line=$first",
  "ppd_last_line=$last",
  "sample_lines_1000=$sample",
  "licensed_incoming=$Incoming",
  "source_gate_ready=$gateReady",
  "has_address_bridge=$hasAddress",
  "has_hmlr_polygon=$hasPolygon",
  "has_title_uprn=$hasTitle",
  "has_combined_bridge=$hasCombined",
  "found_files=$($files.Name -join ',')",
  "overall_progress=$progress",
  "next_action=$next",
  "wait_minutes=$wait",
  "blockers=$($blockers -join ';')",
  "warnings=$($warnings -join ';')",
  "db_write=false",
  "production_deploy=false",
  "fake_data=false"
)
$lines | Set-Content -Encoding UTF8 $FReport
$lines | Set-Content -Encoding UTF8 $GReport
try {
  git -C $Repo add 'docs/chatgpt_status/AUTONOMOUS_PPD_SOURCE_GATE_LATEST.txt'
  git -C $Repo commit -m 'docs: update autonomous PPD source gate result'
  git -C $Repo push origin $Branch
} catch {
  try {
    git -C $Repo fetch origin $Branch
    git -C $Repo pull --rebase origin $Branch
    git -C $Repo push origin $Branch
  } catch {}
}
Write-Host "STATUS=$status"

Write-Host "REPORT=$GReport"
